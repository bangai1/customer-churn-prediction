
import sqlite3
import os

#Data Base path
BASE_DIR = os.path.dirname(os.path.abspath(".."))
DB_PATH = os.path.join(BASE_DIR, "PerfectPetClub.db")

conn = sqlite3.connect(DB_PATH)
cursor = conn.cursor()

cursor.execute("INSERT OR IGNORE INTO product_category (id, name) VALUES (27, 'Subscription');")

# 2. Making the products primary key to 27
cursor.execute("""
UPDATE product 
SET product_category_id = 27 
WHERE name LIKE '%subscription%';
""")

conn.commit()

cursor.execute("SELECT id, name, product_category_id FROM product WHERE name LIKE '%subscription%'")
updated_products = cursor.fetchall()

conn.close()

print("Products in the database")

import subprocess, sys, os

notebooks = [
    "ELT.ipynb",
    "ML.ipynb",
    "Pre_Processed_Data.ipynb"
]

def export_all_to_html(nb_list):
    print(f"Working directory: {os.getcwd()}\n")
    for nb in nb_list:
        if not os.path.exists(nb):
            print(f"[NOT FOUND] {nb}")
            continue
        
        subprocess.run(
            [sys.executable, "-m", "jupyter", "nbconvert", "--to", "html", nb],
            check=True
        )
        html_path = os.path.abspath(nb.replace(".ipynb", ".html"))
        print(f"[OK] {nb} -> {html_path}")

export_all_to_html(notebooks)
